name: Build Android APK

on:
  push:
    branches: [main, master]
  workflow_dispatch:

jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4

      - uses: actions/setup-node@v4
        with:
          node-version: 20

      - uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: 17

      - name: Install dependencies
        run: npm install

      - name: Copy the Excel library into the app
        run: cp node_modules/exceljs/dist/exceljs.min.js www/exceljs.min.js

      - name: Create the Android project
        run: npx cap add android

      - name: Generate app icons
        continue-on-error: true
        run: npx --yes @capacitor/assets generate --android --iconBackgroundColor '#0f4c3a' --iconBackgroundColorDark '#0f4c3a'

      - name: Sync web files into Android
        run: npx cap sync android

      - name: Build the APK
        working-directory: android
        run: |
          chmod +x gradlew
          ./gradlew assembleDebug

      - name: Upload the APK
        uses: actions/upload-artifact@v4
        with:
          name: PSX-Wallet-APK
          path: android/app/build/outputs/apk/debug/app-debug.apk
